     4→import { AnimatePresence, motion } from "framer-motion";
     5→import {
     6→  RotateCcw,

     9→  Dices,

    13→} from "lucide-react";

    17→import { quantumVariants, staggerContainer, viewport } from "@/lib/animations";
    18→import {
    19→  GATES,
    20→  PRESETS,
    21→  ZERO_STATE,
    22→  applyGate,
    23→  bloch,
    24→  cformat,
    25→  cloneState,

    34→/**
    35→ * SECTION 04 — TRY QUANTUM · the interactive qubit playground.
    36→ *
    37→ * A real single-qubit engine (src/lib/quantum.ts) rendered three ways:
    38→ * a Bloch disc, a probability meter, and a circuit wire. The pitch in
    39→ * miniature: "quantum" is not hand-waving — it is this math, and our
    40→ * students will run it on real hardware.
    41→ *
    42→ * RTL note: physics notation (|0⟩, gate symbols, amplitudes) is pinned
    43→ * dir="ltr" inside both languages; only surrounding chrome flips.
    44→ */
    45→
    46→interface Snapshot {
    47→  state: QubitState;
    48→  circuit: string[];
    49→}
    50→
    51→const HISTORY_LIMIT = 32;
    52→const CIRCUIT_WINDOW = 9;
    53→
    54→export default function Playground() {
    55→  const { t, tx, isAr } = useLang();
    56→  const pg = t.playground;
    57→
    58→  const [state, setState] = useState<QubitState>(ZERO_STATE);
    59→  const [circuit, setCircuit] = useState<string[]>([]);
    60→  const [history, setHistory] = useState<Snapshot[]>([]);
    61→  const [flash, setFlash] = useState<0 | 1 | null>(null);
    62→  const [shots, setShots] = useState<{ zeros: number; ones: number } | null>(
    63→    null
    64→  );
    65→  const [liveMsg, setLiveMsg] = useState<string>("");
    66→  const [hoverGate, setHoverGate] = useState<GateDef | null>(null);
    67→
    68→  const { p0, p1 } = useMemo(() => probabilities(state), [state]);
    69→  const { x, z } = useMemo(() => bloch(state), [state]);
    70→  const phase = useMemo(() => relativePhase(state), [state]);
    71→
    72→  /** Push the current configuration onto the decoherence-proof undo stack. */
    73→  const snapshot = useCallback(() => {
    74→    setHistory((prev) => [
    75→      ...prev.slice(-HISTORY_LIMIT + 1),
    76→      { state: cloneState(state), circuit: [...circuit] },
    77→    ]);
    78→  }, [state, circuit]);
    79→
    80→  const applyOne = useCallback(
    81→    (gate: GateDef) => {
    82→      snapshot();
    83→      setState((prev) => applyGate(prev, gate.matrix));
    84→      setCircuit((prev) => [...prev, gate.symbol]);
    85→      setShots(null); // new state → old statistics no longer describe it
    86→      setFlash(null);
    87→    },
    88→    [snapshot]
    89→  );
    90→
    91→  const loadPreset = useCallback(
    92→    (presetState: QubitState) => {
    93→      snapshot();
    94→      setState(cloneState(presetState));
    95→      setCircuit([]);
    96→      setShots(null);
    97→      setFlash(null);
    98→    },
    99→    [snapshot]
   100→  );
   101→
   102→  const undo = useCallback(() => {
   103→    setHistory((prev) => {
   104→      if (prev.length === 0) return prev;
   105→      const last = prev[prev.length - 1];
   106→      setState(last.state);
   107→      setCircuit(last.circuit);
   108→      setShots(null);
   109→      setFlash(null);
   110→      return prev.slice(0, -1);
   111→    });
   112→  }, []);
   113→
   114→  const reset = useCallback(() => {
   115→    snapshot();
   116→    setState(ZERO_STATE);
   117→    setCircuit([]);
   118→    setShots(null);
   119→    setFlash(null);
   120→  }, [snapshot]);
   121→
   122→  /** Projective measurement in the computational basis — collapses ψ. */
   123→  const measureOnce = useCallback(() => {
   124→    snapshot();
   125→    const outcome = sampleOnce(state);
   126→    const collapsed: QubitState =
   127→      outcome === 0
   128→        ? { a: { re: 1, im: 0 }, b: { re: 0, im: 0 } }
   129→        : { a: { re: 0, im: 0 }, b: { re: 1, im: 0 } };
   130→    setState(collapsed);
   131→    setShots(null);
   132→    setFlash(outcome);
   133→    setLiveMsg(pg.collapsedTo(String(outcome)));
   134→  }, [state, snapshot, pg]);
   135→
   136→  /** Born-rule sampling ×100 — the state itself is left untouched. */
   137→  const runShots = useCallback(() => {
   138→    let zeros = 0;
   139→    for (let i = 0; i < 100; i++) if (sampleOnce(state) === 0) zeros++;
   140→    setShots({ zeros, ones: 100 - zeros });
   141→    setLiveMsg(pg.shotsSummary(zeros, 100 - zeros));
   142→  }, [state, pg]);
   143→
   144→  // Bloch disc geometry — a pure state always lies ON the sphere, so the
   145→  // vector length is constant and only its angle ever animates.
   146→  const R = 86;
   147→  const CX = 120;
   148→  const CY = 120;
   149→  const thetaDeg = (Math.atan2(x, z) * 180) / Math.PI;
   150→  const phaseDeg = (phase * 180) / Math.PI;
   151→  const hasB = p1 > 0.001;
   152→
   153→  const visibleChips = circuit.slice(-CIRCUIT_WINDOW);
   154→  const hiddenCount = circuit.length - visibleChips.length;
   155→
   156→  return (
   157→    <section
   158→      id="playground"
   159→      className="relative py-24 md:py-32"
   160→      aria-label={t.misc.playgroundAria}
   161→    >
   162→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   163→        <SectionHeading
   164→          eyebrow={pg.eyebrow}
   165→          title={pg.title}
   166→          subtitle={pg.subtitle}
   167→        />
   168→
   169→        <motion.div
   170→          variants={staggerContainer}
   171→          initial="hidden"
   172→          whileInView="visible"
   173→          viewport={viewport}
   174→          className="grid gap-6 lg:grid-cols-2"
   175→        >
   176→          {/* ── Left: the Bloch disc + probability meter ── */}
   177→          <QuantumCard accent="#00D9FF" noReveal className="p-6 md:p-8">
   178→            <div className="flex flex-col items-center gap-8 sm:flex-row sm:items-start">
   179→              {/* Bloch disc (xz-plane projection of the Bloch sphere) */}
   180→              <div className="relative shrink-0">
   181→                <svg
   182→                  width="240"
   183→                  height="240"
   184→                  viewBox="0 0 240 240"
   185→                  role="img"
   186→                  aria-label={pg.blochAria}
   187→                  dir="ltr"
   188→                  className="drop-shadow-[0_0_24px_rgba(0,217,255,0.15)]"
   189→                >
   190→                  {/* sphere silhouette */}
   191→                  <circle
   192→                    cx={CX}
   193→                    cy={CY}
   194→                    r={R}
   195→                    fill="rgba(17,34,64,0.6)"
   196→                    stroke="rgba(0,217,255,0.35)"
   197→                    strokeWidth="1.5"
   198→                  />
   199→                  {/* equator hint */}
   200→                  <ellipse
   201→                    cx={CX}
   202→                    cy={CY}
   203→                    rx={R}
   204→                    ry={R * 0.32}
   205→                    fill="none"
   206→                    stroke="rgba(108,92,231,0.4)"
   207→                    strokeWidth="1"
   208→                    strokeDasharray="4 5"
   209→                  />
   210→                  {/* z axis */}
   211→                  <line
   212→                    x1={CX}
   213→                    y1={CY - R - 12}
   214→                    x2={CX}
   215→                    y2={CY + R + 12}
   216→                    stroke="rgba(136,146,176,0.35)"
   217→                    strokeWidth="1"
   218→                    strokeDasharray="2 4"
   219→                  />
   220→                  {/* pole labels */}
   221→                  <text
   222→                    x={CX}
   223→                    y={CY - R - 16}
   224→                    textAnchor="middle"
   225→                    fill="#00D9FF"
   226→                    fontSize="14"
   227→                    fontFamily="var(--font-mono), monospace"
   228→                  >
   229→                    |0⟩
   230→                  </text>
   231→                  <text
   232→                    x={CX}
   233→                    y={CY + R + 24}
   234→                    textAnchor="middle"
   235→                    fill="#6C5CE7"
   236→                    fontSize="14"
   237→                    fontFamily="var(--font-mono), monospace"
   238→                  >
   239→                    |1⟩
   240→                  </text>
   241→
   242→                  {/* rotating state vector — length is constant for pure states */}
   243→                  <g transform={`translate(${CX} ${CY})`}>
   244→                    <motion.g
   245→                      animate={{ rotate: thetaDeg }}
   246→                      transition={{ type: "spring", stiffness: 120, damping: 16 }}
   247→                      style={{ transformOrigin: "0px 0px" }}
   248→                    >
   249→                      <line
   250→                        x1="0"
   251→                        y1="0"
   252→                        x2="0"
   253→                        y2={-R}
   254→                        stroke="#00D9FF"
   255→                        strokeWidth="2.5"
   256→                        strokeLinecap="round"
   257→                      />
   258→                      {/* glowing vector tip */}
   259→                      <circle
   260→                        cx="0"
   261→                        cy={-R}
   262→                        r="7"
   263→                        fill="#00D9FF"
   264→                        className="animate-pulse-glow"
   265→                      />
   266→                    </motion.g>
   267→                    {/* origin pivot */}
   268→                    <circle cx="0" cy="0" r="3" fill="#8892B0" />
   269→                  </g>
   270→
   271→                  {/* ground-state shading: |α|² of the disc from the top */}
   272→                  <circle
   273→                    cx={CX}
   274→                    cy={CY}
   275→                    r={R}
   276→                    fill="url(#probGrad)"
   277→                    opacity="0.16"
   278→                    style={{
   279→                      clipPath: `inset(${p0 * 100}% 0 0 0)`,
   280→                    }}
   281→                  />
   282→                  <defs>
   283→                    <linearGradient id="probGrad" x1="0" y1="0" x2="0" y2="1">
   284→                      <stop offset="0%" stopColor="#00D9FF" />
   285→                      <stop offset="100%" stopColor="#6C5CE7" />
   286→                    </linearGradient>
   287→                  </defs>
   288→                </svg>
   289→
   290→                {/* measurement collapse flash */}
   291→                <AnimatePresence>
   292→                  {flash !== null && (
   293→                    <motion.div
   294→                      key={flash}
   295→                      initial={{ opacity: 0, scale: 0.4 }}
   296→                      animate={{ opacity: 1, scale: 1 }}
   297→                      exit={{ opacity: 0, scale: 1.6 }}
   298→                      transition={{ duration: 0.45 }}
   299→                      className="pointer-events-none absolute inset-0 flex items-center justify-center"
   300→                      dir="ltr"
   301→                    >
   302→                      <span className="font-mono text-5xl font-bold text-quantum-blue text-glow">
   303→                        |{flash}⟩
   304→                      </span>
   305→                    </motion.div>
   306→                  )}
   307→                </AnimatePresence>
   308→              </div>
   309→
   310→              {/* probability meter + phase dial + amplitude readout */}
   311→              <div className="flex w-full min-w-0 flex-1 flex-col gap-5">
   312→                <div dir={isAr ? "rtl" : "ltr"}>
   313→                  <p className="mb-2 font-mono text-[10px] tracking-[0.25em] text-quantum-subtle">
   314→                    <Atom className="me-1.5 inline h-3.5 w-3.5 text-quantum-blue" />
   315→                    {pg.stateLabel}
   316→                  </p>
   317→                  <p
   318→                    dir="ltr"
   319→                    className="rounded-lg border border-quantum-blue/15 bg-quantum-navy/70 px-3 py-2.5 text-center font-mono text-sm text-quantum-blue"
   320→                  >
   321→                    |ψ⟩ = {cformat(state.a)}|0⟩ + {cformat(state.b)}|1⟩
   322→                  </p>
   323→                </div>
   324→
   325→                {/* Born-rule probability bars */}
   326→                <div className="space-y-3" role="img" aria-label={`${pg.prob0Label} ${(p0 * 100).toFixed(0)}%, ${pg.prob1Label} ${(p1 * 100).toFixed(0)}%`}>
   327→                  {(
   328→                    [
   329→                      { label: pg.prob0Label, p: p0, color: "#00D9FF" },
   330→                      { label: pg.prob1Label, p: p1, color: "#6C5CE7" },
   331→                    ] as const
   332→                  ).map((row) => (
   333→                    <div key={row.label}>
   334→                      <div className="mb-1 flex items-center justify-between text-[11px] text-quantum-subtle">
   335→                        <span dir={isAr ? "rtl" : "ltr"}>{row.label}</span>
   336→                        <span dir="ltr" className="font-mono text-white">
   337→                          {(row.p * 100).toFixed(1)}%
   338→                        </span>
   339→                      </div>
   340→                      <div className="h-2 overflow-hidden rounded-full bg-quantum-navy/80 ring-1 ring-inset ring-white/5">
   341→                        <motion.div
   342→                          className="h-full rounded-full"
   343→                          style={{
   344→                            background: `linear-gradient(90deg, ${row.color}, ${row.color}99)`,
   345→                            boxShadow: `0 0 12px ${row.color}66`,
   346→                          }}
   347→                          animate={{ width: `${row.p * 100}%` }}
   348→                          transition={{ type: "spring", stiffness: 110, damping: 18 }}
   349→                        />
   350→                      </div>
   351→                    </div>
   352→                  ))}
   353→                </div>
   354→
   355→                {/* relative-phase dial — the invisible axis that makes quantum quantum */}
   356→                <div className="flex items-center gap-4 rounded-lg border border-quantum-purple/20 bg-quantum-navy/60 p-3">
   357→                  <svg width="56" height="56" viewBox="0 0 56 56" dir="ltr" aria-hidden="true">
   358→                    <circle
   359→                      cx="28"
   360→                      cy="28"
   361→                      r="22"
   362→                      fill="none"
   363→                      stroke="rgba(108,92,231,0.35)"
   364→                      strokeWidth="1.5"
   365→                      strokeDasharray="3 4"
   366→                    />
   367→                    <motion.g
   368→                      animate={{ rotate: phaseDeg }}
   369→                      transition={{ type: "spring", stiffness: 120, damping: 16 }}
   370→                      style={{ transformOrigin: "28px 28px" }}
   371→                      opacity={hasB ? 1 : 0.25}
   372→                    >
   373→                      <line
   374→                        x1="28"
   375→                        y1="28"
   376→                        x2="50"
   377→                        y2="28"
   378→                        stroke={hasB ? "#FFD166" : "#8892B0"}
   379→                        strokeWidth="2"
   380→                        strokeLinecap="round"
   381→                      />
   382→                      <circle
   383→                        cx="50"
   384→                        cy="28"
   385→                        r="3"
   386→                        fill={hasB ? "#FFD166" : "#8892B0"}
   387→                      />
   388→                    </motion.g>
   389→                    <circle cx="28" cy="28" r="2.5" fill="#8892B0" />
   390→                  </svg>
   391→                  <div className="min-w-0">
   392→                    <p className="font-mono text-[10px] tracking-[0.2em] text-quantum-subtle" dir={isAr ? "rtl" : "ltr"}>
   393→                      {pg.phaseLabel}
   394→                    </p>
   395→                    <p dir="ltr" className="font-mono text-sm text-white">
   396→                      {hasB ? `${phase.toFixed(2)} rad` : "—"}
   397→                    </p>
   398→                  </div>
   399→                </div>
   400→              </div>
   401→            </div>
   402→          </QuantumCard>
   403→
   404→          {/* ── Right: gate rack, circuit wire, presets, actions ── */}
   405→          <QuantumCard accent="#6C5CE7" noReveal className="flex flex-col gap-6 p-6 md:p-8">
   406→            {/* gates */}
   407→            <div aria-label={pg.controlsAria}>
   408→              <div className="mb-3 flex items-baseline justify-between gap-3">
   409→                <h3 className="flex items-center gap-2 font-heading text-sm font-bold text-white">
   410→                  <Gauge className="h-4 w-4 text-quantum-purple" />
   411→                  {pg.gatesTitle}
   412→                </h3>
   413→                <p className="text-[11px] text-quantum-subtle">{pg.gatesHint}</p>
   414→              </div>
   415→              <div className="grid grid-cols-5 gap-2">
   416→                {GATES.map((gate) => (
   417→                  <button
   418→                    key={gate.symbol}
   419→                    type="button"
   420→                    onClick={() => applyOne(gate)}
   421→                    onMouseEnter={() => setHoverGate(gate)}
   422→                    onMouseLeave={() => setHoverGate(null)}
   423→                    onFocus={() => setHoverGate(gate)}
   424→                    onBlur={() => setHoverGate(null)}
   425→                    aria-label={pg.gateAria(tx(gate.name))}
   426→                    aria-describedby="gate-blurb"
   427→                    className="group/gate flex h-16 flex-col items-center justify-center gap-1 rounded-xl border border-white/10 bg-quantum-navy/70 transition-all duration-200 hover:-translate-y-0.5 hover:border-[color-mix(in_srgb,var(--gate-accent)_60%,transparent)] focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-quantum-blue"
   428→                    style={{ "--gate-accent": gate.accent } as React.CSSProperties}
   429→                  >
   430→                    <span
   431→                      className="font-mono text-lg font-bold transition-colors"
   432→                      style={{ color: gate.accent }}
   433→                    >
   434→                      {gate.symbol}
   435→                    </span>
   436→                    <span className="px-1 text-center text-[9px] leading-tight text-quantum-subtle">
   437→                      {tx(gate.name)}
   438→                    </span>
   439→                  </button>
   440→                ))}
   441→              </div>
   442→              {/* gate intuition panel (hover/focus driven) */}
   443→              <div
   444→                id="gate-blurb"
   445→                aria-live="polite"
   446→                className="mt-3 min-h-[3.25rem] rounded-lg border border-white/5 bg-quantum-navy/60 px-3 py-2.5 text-xs leading-relaxed text-quantum-subtle"
   447→              >
   448→                {hoverGate ? (
   449→                  <span>
   450→                    <span
   451→                      className="me-1.5 font-mono font-bold"
   452→                      style={{ color: hoverGate.accent }}
   453→                      dir="ltr"
   454→                    >
   455→                      {hoverGate.symbol}
   456→                    </span>
   457→                    {tx(hoverGate.blurb)}
   458→                  </span>
   459→                ) : (
   460→                  <span className="opacity-50" dir={isAr ? "rtl" : "ltr"}>
   461→                    {pg.gatesHint}
   462→                  </span>
   463→                )}
   464→              </div>
   465→            </div>
   466→
   467→            {/* circuit wire */}
   468→            <div>
   469→              <h3 className="mb-3 flex items-center gap-2 font-heading text-sm font-bold text-white">
   470→                <CircuitBoard className="h-4 w-4 text-quantum-blue" />
   471→                {pg.circuitTitle}
   472→                {circuit.length > 0 && (
   473→                  <span className="ms-auto font-mono text-[10px] text-quantum-subtle" dir="ltr">
   474→                    {circuit.length} op{circuit.length === 1 ? "" : "s"}
   475→                  </span>
   476→                )}
   477→              </h3>
   478→              <div
   479→                dir="ltr"
   480→                className="relative flex min-h-[3.5rem] items-center overflow-hidden rounded-lg border border-quantum-blue/15 bg-quantum-navy/70 px-3"
   481→              >
   482→                {/* the wire */}
   483→                <div
   484→                  className="pointer-events-none absolute inset-x-3 top-1/2 h-px bg-gradient-to-r from-transparent via-quantum-blue/40 to-transparent"
   485→                  aria-hidden="true"
   486→                />
   487→                {visibleChips.length === 0 ? (
   488→                  <span className="relative z-10 w-full text-center font-mono text-xs text-quantum-subtle/70">
   489→                    {pg.emptyCircuit}
   490→                  </span>
   491→                ) : (
   492→                  <div className="relative z-10 flex w-full items-center gap-1.5 py-2">
   493→                    {hiddenCount > 0 && (
   494→                      <span className="font-mono text-[10px] text-quantum-subtle">
   495→                        +{hiddenCount}
   496→                      </span>
   497→                    )}
   498→                    <AnimatePresence initial={false}>
   499→                      {visibleChips.map((symbol, idx) => {
   500→                        const gate = GATES.find((g) => g.symbol === symbol);
   501→                        return (
   502→                          <motion.span
   503→                            key={`${idx}-${symbol}`}
   504→                            initial={{ scale: 0, opacity: 0 }}
   505→                            animate={{ scale: 1, opacity: 1 }}
   506→                            transition={{ type: "spring", stiffness: 300, damping: 20 }}
   507→                            className="flex h-9 w-9 shrink-0 items-center justify-center rounded-md border bg-quantum-secondary font-mono text-sm font-bold shadow-[0_0_10px_rgba(0,0,0,0.4)]"
   508→                            style={{
   509→                              color: gate?.accent ?? "#00D9FF",
   510→                              borderColor: `${gate?.accent ?? "#00D9FF"}55`,
   511→                            }}
   512→                          >
   513→                            {symbol}
   514→                          </motion.span>
   515→                        );
   516→                      })}
   517→                    </AnimatePresence>
   518→                    {/* metering dash at the end of the wire */}
   519→                    <motion.span
   520→                      className="ms-auto flex h-9 w-9 shrink-0 items-center justify-center rounded-md border border-quantum-blue/40 bg-quantum-navy font-mono text-xs text-quantum-blue"
   521→                      animate={{ boxShadow: [
   522→                        "0 0 0 rgba(0,217,255,0)",
   523→                        "0 0 14px rgba(0,217,255,0.45)",
   524→                        "0 0 0 rgba(0,217,255,0)",
   525→                      ] }}
   526→                      transition={{ duration: 2.4, repeat: Infinity }}
   527→                      title="measure"
   528→                    >
   529→                      ⌖
   530→                    </motion.span>
   531→                  </div>
   532→                )}
   533→              </div>
   534→            </div>
   535→
   536→            {/* presets */}
   537→            <div>
   538→              <h3 className="mb-3 flex items-center gap-2 font-heading text-sm font-bold text-white">
   539→                <Orbit className="h-4 w-4 text-quantum-green" />
   540→                {pg.presetsTitle}
   541→              </h3>
   542→              <div className="flex flex-wrap gap-2">
   543→                {PRESETS.map((preset) => (
   544→                  <button
   545→                    key={preset.label}
   546→                    type="button"
   547→                    onClick={() => loadPreset(preset.state)}
