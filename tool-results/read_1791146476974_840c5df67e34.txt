    36→/**
    37→ * The two-qubit wing of SECTION 04 — TRY QUANTUM.
    38→ *
    39→ * One qubit showed superposition; two qubits show the thing that actually
    40→ * gives quantum computers their power: ENTANGLEMENT. The centerpiece is
    41→ * CNOT + H = Bell pair, and the signature visual is the two Bloch discs
    42→ * shrinking to their centers as the entanglement meter fills — each qubit
    43→ * alone becomes pure noise while the correlation becomes perfect.
    44→ *
    45→ * Physics honesty notes:
    46→ *  - Each disc shows the reduced (mixed) state, so the vector LENGTH now
    47→ *    varies: length 1 = pure/separable, length 0 = maximally mixed.
    48→ *  - The meter is the concurrence C = 2|c₀c₃ − c₁c₂| — exact for pure states.
    49→ *  - Everything runs at build-free display precision, renormalized per gate.
    50→ *
    51→ * RTL note: kets, circuit lanes and C values stay dir="ltr"; chrome flips.
    52→ */
    53→
    54→const HISTORY_LIMIT = 32;
    55→const CIRCUIT_WINDOW = 6;

    66→  id: string;
    67→  label: string;
    68→  accent: string;

    78→/**
    79→ * Mini Bloch disc of ONE qubit's reduced state. Unlike the single-qubit
    80→ * disc, the vector length varies — watching it shrink toward the center
    81→ * as CNOT entangles the pair is the whole lesson in one picture.
    82→ */
    83→function ReducedBloch({
    84→  x,
    85→  z,
    86→  qubit,
    87→  ariaLabel,
    88→}: {
    89→  x: number;
    90→  z: number;
    91→  qubit: 0 | 1;
    92→  ariaLabel: string;
    93→}) {
    94→  const SIZE = 150;
    95→  const R = SIZE / 2 - 20;
    96→  const C = SIZE / 2;
    97→  const len = Math.min(1, Math.hypot(x, z)); // |ρ| ∈ [0,1]
    98→  const thetaDeg = (Math.atan2(x, z) * 180) / Math.PI;
    99→  const mixed = len < 0.05; // maximally mixed → pure noise, no direction
   100→  const accent = qubit === 0 ? "#00D9FF" : "#6C5CE7";
   101→
   102→  return (
   103→    <svg
   104→      width={SIZE}
   105→      height={SIZE}
   106→      viewBox={`0 0 ${SIZE} ${SIZE}`}
   107→      role="img"
   108→      aria-label={ariaLabel}
   109→      dir="ltr"
   110→      className="drop-shadow-[0_0_18px_rgba(0,217,255,0.12)]"
   111→    >
   112→      {/* sphere silhouette */}
   113→      <circle cx={C} cy={C} r={R} fill="rgba(17,34,64,0.6)" stroke="rgba(136,146,176,0.3)" strokeWidth="1.2" />
   114→      {/* equator hint */}
   115→      <ellipse cx={C} cy={C} rx={R} ry={R * 0.32} fill="none" stroke="rgba(136,146,176,0.25)" strokeWidth="1" strokeDasharray="3 4" />
   116→      {/* poles */}
   117→      <text x={C} y={C - R - 8} textAnchor="middle" fill="#8892B0" fontSize="11" fontFamily="var(--font-mono), monospace">
   118→        |0⟩
   119→      </text>
   120→      <text x={C} y={C + R + 16} textAnchor="middle" fill="#8892B0" fontSize="11" fontFamily="var(--font-mono), monospace">
   121→        |1⟩
   122→      </text>
   123→
   124→      {mixed ? (
   125→        /* maximally mixed: no direction at all — render a breathing blob */
   126→        <motion.circle
   127→          cx={C}
   128→          cy={C}
   129→          r={6}
   130→          fill={accent}
   131→          animate={{ opacity: [0.35, 0.9, 0.35], scale: [0.8, 1.25, 0.8] }}
   132→          transition={{ duration: 2, repeat: Infinity, ease: "easeInOut" }}
   133→          style={{ transformOrigin: `${C}px ${C}px` }}
   134→        />
   135→      ) : (
   136→        <g transform={`translate(${C} ${C})`}>
   137→          <motion.g
   138→            animate={{ rotate: thetaDeg, scale: len }}
   139→            transition={{ type: "spring", stiffness: 110, damping: 16 }}
   140→            style={{ transformOrigin: "0px 0px" }}
   141→          >
   142→            <line x1="0" y1="0" x2="0" y2={-R} stroke={accent} strokeWidth="2.5" strokeLinecap="round" />
   143→            <circle cx="0" cy={-R} r="6" fill={accent} className="animate-pulse-glow" />
   144→          </motion.g>
   145→          <circle cx="0" cy="0" r="2.5" fill="#8892B0" />
   146→        </g>
   147→      )}
   148→    </svg>
   149→  );
   150→}

   156→const LANE_TOP = 18; // px — center of q0 lane inside a 72px column
   157→const LANE_BOTTOM = 54; // px — center of q1 lane
   158→
   159→/** One column of the two-lane circuit: a gate chip or a CNOT pair. */
   160→function OpColumn({ op }: { op: TwoQubitOp }) {
   161→  if (op.kind === "gate") {
   162→    const gate = GATES.find((g) => g.symbol === op.symbol);
   163→    const accent = gate?.accent ?? "#00D9FF";
   164→    return (
   165→      <motion.span
   166→        initial={{ scale: 0, opacity: 0 }}
   167→        animate={{ scale: 1, opacity: 1 }}
   168→        transition={{ type: "spring", stiffness: 300, damping: 20 }}
   169→        className="relative block h-[72px] w-10 shrink-0"
   170→      >
   171→        <span
   172→          className="absolute flex h-9 w-9 items-center justify-center rounded-md border bg-quantum-secondary font-mono text-sm font-bold shadow-[0_0_10px_rgba(0,0,0,0.4)]"
   173→          style={{
   174→            color: accent,
   175→            borderColor: `${accent}55`,
   176→            top: op.target === 0 ? LANE_TOP - 18 : LANE_BOTTOM - 18,
   177→            left: "50%",
   178→            transform: "translateX(-50%)",
   179→          }}
   180→        >
   181→          {op.symbol}
   182→        </span>
   183→      </motion.span>
   184→    );
   185→  }
   186→
   187→  // CNOT: control dot on one lane, ⊕ target on the other, vertical connector.
   188→  const ctrlY = op.control === 0 ? LANE_TOP : LANE_BOTTOM;
   189→  const tgtY = op.control === 0 ? LANE_BOTTOM : LANE_TOP;
   190→  return (
   191→    <motion.span
   192→      initial={{ scale: 0, opacity: 0 }}
   193→      animate={{ scale: 1, opacity: 1 }}
   194→      transition={{ type: "spring", stiffness: 300, damping: 20 }}
   195→      className="relative block h-[72px] w-10 shrink-0"
   196→    >
   197→      {/* vertical connector */}
   198→      <span
   199→        className="absolute left-1/2 w-0.5 -translate-x-1/2 bg-quantum-green shadow-[0_0_8px_rgba(0,184,148,0.6)]"
   200→        style={{ top: Math.min(ctrlY, tgtY), height: Math.abs(ctrlY - tgtY) }}
   201→        aria-hidden="true"
   202→      />
   203→      {/* control dot */}
   204→      <span
   205→        className="absolute left-1/2 h-3 w-3 -translate-x-1/2 -translate-y-1/2 rounded-full bg-quantum-green shadow-[0_0_10px_rgba(0,184,148,0.8)]"
   206→        style={{ top: ctrlY }}
   207→        aria-hidden="true"
   208→      />
   209→      {/* target ⊕ */}
   210→      <span
   211→        className="absolute left-1/2 flex h-5 w-5 -translate-x-1/2 -translate-y-1/2 items-center justify-center rounded-full border-2 border-quantum-green bg-quantum-navy font-mono text-[11px] font-bold leading-none text-quantum-green"
   212→        style={{ top: tgtY }}
   213→        aria-hidden="true"
   214→      >
   215→        +
   216→      </span>
   217→    </motion.span>
   218→  );
   219→}

   225→export default function TwoQubitLab() {
   226→  const { t, tx, isAr } = useLang();
   227→  const pg = t.playground;
   228→  const tq = t.twoQubit;
   229→
   230→  const [state, setState] = useState<TwoQubitState>(TWO_ZERO_STATE);
   231→  const [circuit, setCircuit] = useState<TwoQubitOp[]>([]);
   232→  const [history, setHistory] = useState<Snapshot[]>([]);
   233→  const [flash, setFlash] = useState<number | null>(null);
   234→  const [shots, setShots] = useState<[number, number, number, number] | null>(null);
   235→  const [liveMsg, setLiveMsg] = useState<string>("");
   236→  const [hoverOp, setHoverOp] = useState<RackButton | null>(null);
   237→  const [celebrated, setCelebrated] = useState(false);
   238→
   239→  const probs = useMemo(() => twoProbabilities(state), [state]);
   240→  const C = useMemo(() => concurrence(state), [state]);
   241→  const blochQ0 = useMemo(() => reducedBloch(state, 0), [state]);
   242→  const blochQ1 = useMemo(() => reducedBloch(state, 1), [state]);
   243→  const isBell = C > 0.99;
   244→
   245→  const H_MATRIX = GATES[0].matrix;
   246→  const X_MATRIX = GATES[1].matrix;
   247→
   248→  /** The six-operation rack: 4 single-wire gates + 2 CNOT directions. */
   249→  const RACK: RackButton[] = [
   250→    { id: "h0", label: "H·q0", accent: "#00D9FF", blurb: tq.blurbH(0), apply: (s) => applySingleToWire(s, H_MATRIX, 0), op: { kind: "gate", symbol: "H", target: 0 } },
   251→    { id: "x0", label: "X·q0", accent: "#6C5CE7", blurb: tq.blurbX(0), apply: (s) => applySingleToWire(s, X_MATRIX, 0), op: { kind: "gate", symbol: "X", target: 0 } },
   252→    { id: "h1", label: "H·q1", accent: "#00D9FF", blurb: tq.blurbH(1), apply: (s) => applySingleToWire(s, H_MATRIX, 1), op: { kind: "gate", symbol: "H", target: 1 } },
   253→    { id: "x1", label: "X·q1", accent: "#6C5CE7", blurb: tq.blurbX(1), apply: (s) => applySingleToWire(s, X_MATRIX, 1), op: { kind: "gate", symbol: "X", target: 1 } },
   254→    { id: "c01", label: "⊕ 0→1", accent: "#00B894", blurb: tq.blurbCnot(0, 1), apply: (s) => applyCNOT(s, 0), op: { kind: "cnot", control: 0 } },
   255→    { id: "c10", label: "⊕ 1→0", accent: "#00B894", blurb: tq.blurbCnot(1, 0), apply: (s) => applyCNOT(s, 1), op: { kind: "cnot", control: 1 } },
   256→  ];
   257→
   258→  const snapshot = useCallback(() => {
   259→    setHistory((prev) => [
   260→      ...prev.slice(-HISTORY_LIMIT + 1),
   261→      { state: cloneTwoState(state), circuit: [...circuit] },
   262→    ]);
   263→  }, [state, circuit]);
   264→
   265→  const clearTransient = () => {
   266→    setShots(null);
   267→    setFlash(null);
   268→  };
   269→
   270→  const applyOne = useCallback(
   271→    (btn: RackButton) => {
   272→      snapshot();
   273→      const next = btn.apply(state);
   274→      setState(next);
   275→      setCircuit((prev) => [...prev, btn.op]);
   276→      clearTransient();
   277→      // One-shot celebration the first time a Bell pair appears.
   278→      if (!celebrated && concurrence(next) > 0.99) setCelebrated(true);
   279→    },
   280→    [snapshot, state, celebrated]
   281→  );
   282→
   283→  const loadPreset = useCallback(
   284→    (presetState: TwoQubitState) => {
   285→      snapshot();
   286→      setState(cloneTwoState(presetState));
   287→      setCircuit([]);
   288→      clearTransient();
   289→    },
   290→    [snapshot]
   291→  );
   292→
   293→  const undo = useCallback(() => {
   294→    setHistory((prev) => {
   295→      if (prev.length === 0) return prev;
   296→      const last = prev[prev.length - 1];
   297→      setState(last.state);
   298→      setCircuit(last.circuit);
   299→      clearTransient();
   300→      return prev.slice(0, -1);
   301→    });
   302→  }, []);
   303→
   304→  const reset = useCallback(() => {
   305→    snapshot();
   306→    setState(TWO_ZERO_STATE);
   307→    setCircuit([]);
   308→    clearTransient();
   309→  }, [snapshot]);
   310→
   311→  /** Destructive projective measurement of BOTH qubits. */
   312→  const measureOnce = useCallback(() => {
   313→    snapshot();
   314→    const outcome = sampleTwoOnce(state);
   315→    setState(collapseTwo(outcome));
   316→    clearTransient();
   317→    const label = ketLabel(outcome);
   318→    setLiveMsg(tq.collapsedTo2(label));
   319→  }, [state, snapshot, tq]);
   320→
   321→  /** Born-rule sampling ×100 — non-destructive. */
   322→  const runShots = useCallback(() => {
   323→    const counts: [number, number, number, number] = [0, 0, 0, 0];
   324→    for (let i = 0; i < 100; i++) counts[sampleTwoOnce(state)]++;
   325→    setShots(counts);
   326→    const summary = counts
   327→      .map((n, i) => `${n}× ${ketLabel(i)}`)
   328→      .filter((s) => !s.startsWith("0×"))
   329→      .join(" · ");
   330→    setLiveMsg(tq.shotsSummary2(summary));
   331→  }, [state, tq]);
   332→
   333→  const visibleOps = circuit.slice(-CIRCUIT_WINDOW);
   334→  const hiddenCount = circuit.length - visibleOps.length;
   335→
   336→  const jointAria = probs
   337→    .map((p, i) => `${ketLabel(i)} ${(p * 100).toFixed(0)}%`)
   338→    .join(", ");
   339→
   340→  return (
   341→    <motion.div
   342→      variants={staggerContainer}
   343→      initial="hidden"
   344→      whileInView="visible"
   345→      viewport={viewport}
   346→      className="grid gap-6 lg:grid-cols-2"
   347→    >
   348→      {/* ── Left: reduced Bloch discs + entanglement meter + joint probs ── */}
   349→      <QuantumCard accent={ENTANGLED_COLOR} noReveal className="p-6 md:p-8">
   350→        <div className="mb-1 flex items-baseline justify-between gap-3">
   351→          <h3 className="flex items-center gap-2 font-heading text-sm font-bold text-white">
   352→            <Orbit className="h-4 w-4 text-quantum-purple" />
   353→            {tq.spheresTitle}
   354→          </h3>
   355→        </div>
   356→        <p className="mb-5 text-[11px] text-quantum-subtle">{tq.wireHint}</p>
   357→
   358→        {/* the pair, linked by their correlation */}
   359→        <div className="relative flex items-center justify-center gap-1 sm:gap-3">
   360→          <div className="flex flex-col items-center">
   361→            <ReducedBloch
   362→              x={blochQ0.x}
   363→              z={blochQ0.z}
   364→              qubit={0}
   365→              ariaLabel={tq.reducedAria(0)}
   366→            />
   367→            <p className="mt-1 font-mono text-[10px] tracking-wider text-quantum-subtle" dir="ltr">
   368→              {tq.qubitLabel(0)}
   369→            </p>
   370→          </div>
   371→
   372→          {/* correlation beam — intensity IS the concurrence */}
   373→          <div className="flex w-10 flex-col items-center gap-1.5 sm:w-14" aria-hidden="true">
   374→            <span
   375→              className="font-mono text-lg transition-colors"
   376→              style={{ color: ENTANGLED_COLOR, opacity: 0.35 + C * 0.65 }}
   377→            >
   378→              ⊗
   379→            </span>
   380→            <div
   381→              className={`h-16 w-1 rounded-full ${isBell ? "animate-pulse-glow" : ""}`}
   382→              style={{
   383→                background: `linear-gradient(180deg, transparent, ${ENTANGLED_COLOR}, transparent)`,
   384→                opacity: 0.12 + C * 0.88,
   385→                boxShadow: C > 0.9 ? `0 0 18px ${ENTANGLED_COLOR}` : "none",
   386→              }}
   387→            />
   388→            <span className="font-mono text-[10px] text-quantum-subtle" dir="ltr">
   389→              C={C.toFixed(2)}
   390→            </span>
   391→          </div>
   392→
   393→          <div className="flex flex-col items-center">
   394→            <ReducedBloch

   396→              z={blochQ1.z}

   399→            />
   400→            <p className="mt-1 font-mono text-[10px] tracking-wider text-quantum-subtle" dir="ltr">

   405→          {/* collapse flash — shared by the pair */}

   414→                className="pointer-events-none absolute inset-0 z-10 flex items-center justify-center"

   424→
   425→        {/* entanglement meter */}
   426→        <div className="mt-6" role="img" aria-label={tq.meterAria(C)}>
   427→          <div className="mb-1.5 flex items-center justify-between">
   428→            <span className="flex items-center gap-1.5 font-mono text-[10px] tracking-[0.25em] text-quantum-subtle">
   429→              <Link2 className="h-3.5 w-3.5" style={{ color: ENTANGLED_COLOR }} />
   430→              {tq.meterLabel}
   431→            </span>
   432→            <span className="font-mono text-sm text-white" dir="ltr">

   441→                boxShadow: `0 0 14px ${ENTANGLED_COLOR}66`,
   442→              }}
   443→              animate={{ width: `${C * 100}%` }}
   444→              transition={{ type: "spring", stiffness: 110, damping: 18 }}
   445→            />
   446→          </div>
   447→          <div className="mt-1 flex justify-between text-[10px] text-quantum-subtle/80">

   453→        {/* Bell achievement — the payoff moment */}

   463→              <p className="flex items-center gap-2 font-heading text-sm font-bold text-[#FF6B9D]">
