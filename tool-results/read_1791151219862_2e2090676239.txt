    35→    35→
    36→    36→function SignalCounter() {
    37→    37→  const [total, setTotal] = useState<number | null>(null);
    38→    38→  const { t } = useLang();
    39→    39→
    40→    40→  useEffect(() => {
    41→    41→    let cancelled = false;
    42→    42→    fetch("/api/inquiries")
    43→    43→      .then((r) => r.json())
    44→    44→      .then((data: { ok: boolean; total?: number }) => {
    45→    45→        if (!cancelled && data.ok && typeof data.total === "number") {
    46→    46→          setTotal(data.total);
    47→    47→        }
    48→    48→      })
    49→    49→      .catch(() => {
    50→    50→        /* the counter stays in superposition */
    51→    51→      });
    52→    52→    return () => {
    53→    53→      cancelled = true;
    54→    54→    };
    55→    55→  }, []);
    56→    56→
    57→    57→  return (
    58→    58→    <div className="flex items-center justify-center gap-2.5 rounded-full border border-quantum-green/30 bg-quantum-green/10 px-5 py-2.5 mb-10 w-fit mx-auto">
    59→    59→      <motion.span
    60→    60→        animate={{ opacity: [1, 0.35, 1] }}
    61→    61→        transition={{ duration: 1.6, repeat: Infinity }}
    62→    62→        className="relative flex size-2.5"
    63→    63→      >
    64→    64→        <span className="absolute inline-flex size-full animate-ping rounded-full bg-quantum-green opacity-60" />
    65→    65→        <span className="relative inline-flex size-2.5 rounded-full bg-quantum-green" />
    66→    66→      </motion.span>
    67→    67→      <span className="font-mono text-xs tracking-wider text-quantum-green">
    68→    68→        {total === null ? t.contact.counterScanning : t.contact.counter(total)}
    69→    69→      </span>
    70→    70→    </div>
    71→    71→  );
    72→    72→}
