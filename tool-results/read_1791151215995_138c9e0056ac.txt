    31→ * A fullstack form: signals POST to /api/inquiries (Prisma + SQLite),
    32→ * and a live counter shows how many "measurements" have collapsed so far.
    33→ * Validation messages re-bind per language — the schema is rebuilt on toggle.

    35→
    36→function SignalCounter() {
    37→  const [total, setTotal] = useState<number | null>(null);
    38→  const { t } = useLang();
    39→
    40→  useEffect(() => {
    41→    let cancelled = false;
    42→    fetch("/api/inquiries")
    43→      .then((r) => r.json())
    44→      .then((data: { ok: boolean; total?: number }) => {
    45→        if (!cancelled && data.ok && typeof data.total === "number") {
    46→          setTotal(data.total);
    47→        }
    48→      })
    49→      .catch(() => {
    50→        /* the counter stays in superposition */
    51→      });
    52→    return () => {
    53→      cancelled = true;
    54→    };
    55→  }, []);
    56→
    57→  return (
    58→    <div className="flex items-center justify-center gap-2.5 rounded-full border border-quantum-green/30 bg-quantum-green/10 px-5 py-2.5 mb-10 w-fit mx-auto">
    59→      <motion.span
    60→        animate={{ opacity: [1, 0.35, 1] }}
    61→        transition={{ duration: 1.6, repeat: Infinity }}
    62→        className="relative flex size-2.5"
    63→      >
    64→        <span className="absolute inline-flex size-full animate-ping rounded-full bg-quantum-green opacity-60" />
    65→        <span className="relative inline-flex size-2.5 rounded-full bg-quantum-green" />
    66→      </motion.span>
    67→      <span className="font-mono text-xs tracking-wider text-quantum-green">
    68→        {total === null ? t.contact.counterScanning : t.contact.counter(total)}
    69→      </span>
    70→    </div>
    71→  );
    72→}
