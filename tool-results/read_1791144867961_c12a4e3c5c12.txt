    34→    34→/**
    35→    35→ * SECTION 04 — TRY QUANTUM · the interactive qubit playground.
    36→    36→ *
    37→    37→ * A real single-qubit engine (src/lib/quantum.ts) rendered three ways:
    38→    38→ * a Bloch disc, a probability meter, and a circuit wire. The pitch in
    39→    39→ * miniature: "quantum" is not hand-waving — it is this math, and our
    40→    40→ * students will run it on real hardware.
    41→    41→ *
    42→    42→ * RTL note: physics notation (|0⟩, gate symbols, amplitudes) is pinned
    43→    43→ * dir="ltr" inside both languages; only surrounding chrome flips.
    44→    44→ */
    45→    45→
    46→    46→interface Snapshot {
    47→    47→  state: QubitState;
    48→    48→  circuit: string[];
    49→    49→}
    50→    50→
    51→    51→const HISTORY_LIMIT = 32;
    52→    52→const CIRCUIT_WINDOW = 9;
    53→    53→
    54→    54→export default function Playground() {
    55→    55→  const { t, tx, isAr } = useLang();
    56→    56→  const pg = t.playground;
    57→    57→
    58→    58→  const [state, setState] = useState<QubitState>(ZERO_STATE);
    59→    59→  const [circuit, setCircuit] = useState<string[]>([]);
    60→    60→  const [history, setHistory] = useState<Snapshot[]>([]);
    61→    61→  const [flash, setFlash] = useState<0 | 1 | null>(null);
    62→    62→  const [shots, setShots] = useState<{ zeros: number; ones: number } | null>(
    63→    63→    null
    64→    64→  );
    65→    65→  const [liveMsg, setLiveMsg] = useState<string>("");
    66→    66→  const [hoverGate, setHoverGate] = useState<GateDef | null>(null);
    67→    67→
    68→    68→  const { p0, p1 } = useMemo(() => probabilities(state), [state]);
    69→    69→  const { x, z } = useMemo(() => bloch(state), [state]);
    70→    70→  const phase = useMemo(() => relativePhase(state), [state]);
    71→    71→
    72→    72→  /** Push the current configuration onto the decoherence-proof undo stack. */
    73→    73→  const snapshot = useCallback(() => {
    74→    74→    setHistory((prev) => [
    75→    75→      ...prev.slice(-HISTORY_LIMIT + 1),
    76→    76→      { state: cloneState(state), circuit: [...circuit] },
    77→    77→    ]);
    78→    78→  }, [state, circuit]);
    79→    79→
    80→    80→  const applyOne = useCallback(
    81→    81→    (gate: GateDef) => {
    82→    82→      snapshot();
    83→    83→      setState((prev) => applyGate(prev, gate.matrix));
    84→    84→      setCircuit((prev) => [...prev, gate.symbol]);
    85→    85→      setShots(null); // new state → old statistics no longer describe it
    86→    86→      setFlash(null);
    87→    87→    },
    88→    88→    [snapshot]
    89→    89→  );
    90→    90→
    91→    91→  const loadPreset = useCallback(
    92→    92→    (presetState: QubitState) => {
    93→    93→      snapshot();
    94→    94→      setState(cloneState(presetState));
    95→    95→      setCircuit([]);
    96→    96→      setShots(null);
    97→    97→      setFlash(null);
    98→    98→    },
    99→    99→    [snapshot]
   100→   100→  );
   101→   101→
   102→   102→  const undo = useCallback(() => {
   103→   103→    setHistory((prev) => {
   104→   104→      if (prev.length === 0) return prev;
   105→   105→      const last = prev[prev.length - 1];
   106→   106→      setState(last.state);
   107→   107→      setCircuit(last.circuit);
   108→   108→      setShots(null);
   109→   109→      setFlash(null);
   110→   110→      return prev.slice(0, -1);
   111→   111→    });
   112→   112→  }, []);
   113→   113→
   114→   114→  const reset = useCallback(() => {
   115→   115→    snapshot();
   116→   116→    setState(ZERO_STATE);
   117→   117→    setCircuit([]);
   118→   118→    setShots(null);
   119→   119→    setFlash(null);
   120→   120→  }, [snapshot]);
   121→   121→
   122→   122→  /** Projective measurement in the computational basis — collapses ψ. */
   123→   123→  const measureOnce = useCallback(() => {
   124→   124→    snapshot();
   125→   125→    const outcome = sampleOnce(state);
   126→   126→    const collapsed: QubitState =
   127→   127→      outcome === 0
   128→   128→        ? { a: { re: 1, im: 0 }, b: { re: 0, im: 0 } }
   129→   129→        : { a: { re: 0, im: 0 }, b: { re: 1, im: 0 } };
   130→   130→    setState(collapsed);
   131→   131→    setShots(null);
   132→   132→    setFlash(outcome);
   133→   133→    setLiveMsg(pg.collapsedTo(String(outcome)));
   134→   134→  }, [state, snapshot, pg]);
   135→   135→
   136→   136→  /** Born-rule sampling ×100 — the state itself is left untouched. */
   137→   137→  const runShots = useCallback(() => {
   138→   138→    let zeros = 0;
   139→   139→    for (let i = 0; i < 100; i++) if (sampleOnce(state) === 0) zeros++;
   140→   140→    setShots({ zeros, ones: 100 - zeros });
   141→   141→    setLiveMsg(pg.shotsSummary(zeros, 100 - zeros));
   142→   142→  }, [state, pg]);
   143→   143→
   144→   144→  // Bloch disc geometry — a pure state always lies ON the sphere, so the
   145→   145→  // vector length is constant and only its angle ever animates.
   146→   146→  const R = 86;
   147→   147→  const CX = 120;
   148→   148→  const CY = 120;
   149→   149→  const thetaDeg = (Math.atan2(x, z) * 180) / Math.PI;
   150→   150→  const phaseDeg = (phase * 180) / Math.PI;
   151→   151→  const hasB = p1 > 0.001;
   152→   152→
   153→   153→  const visibleChips = circuit.slice(-CIRCUIT_WINDOW);
   154→   154→  const hiddenCount = circuit.length - visibleChips.length;

   157→   157→    <section
   158→   158→      id="playground"
   159→   159→      className="relative py-24 md:py-32"
   160→   160→      aria-label={t.misc.playgroundAria}
   161→   161→    >
   162→   162→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   163→   163→        <SectionHeading
   164→   164→          eyebrow={pg.eyebrow}
   165→   165→          title={pg.title}
   166→   166→          subtitle={pg.subtitle}
   167→   167→        />
   168→   168→
   169→   169→        <motion.div
   170→   170→          variants={staggerContainer}

   173→   173→          viewport={viewport}
   174→   174→          className="grid gap-6 lg:grid-cols-2"
   175→   175→        >

   177→   177→          <QuantumCard accent="#00D9FF" noReveal className="p-6 md:p-8">
   178→   178→            <div className="flex flex-col items-center gap-8 sm:flex-row sm:items-start">
   179→   179→              {/* Bloch disc (xz-plane projection of the Bloch sphere) */}
   180→   180→              <div className="relative shrink-0">
   181→   181→                <svg
   182→   182→                  width="240"
   183→   183→                  height="240"
   184→   184→                  viewBox="0 0 240 240"
   185→   185→                  role="img"
   186→   186→                  aria-label={pg.blochAria}
   187→   187→                  dir="ltr"
   188→   188→                  className="drop-shadow-[0_0_24px_rgba(0,217,255,0.15)]"
   189→   189→                >
   190→   190→                  {/* sphere silhouette */}
   191→   191→                  <circle
   192→   192→                    cx={CX}
   193→   193→                    cy={CY}
   194→   194→                    r={R}
   195→   195→                    fill="rgba(17,34,64,0.6)"
   196→   196→                    stroke="rgba(0,217,255,0.35)"
   197→   197→                    strokeWidth="1.5"

   201→   201→                    cx={CX}

   203→   203→                    rx={R}
   204→   204→                    ry={R * 0.32}

   208→   208→                    strokeDasharray="4 5"

   300→   300→                      dir="ltr"

   302→   302→                      <span className="font-mono text-5xl font-bold text-quantum-blue text-glow">

   304→   304→                      </span>
   305→   305→                    </motion.div>
   306→   306→                  )}
   307→   307→                </AnimatePresence>
   308→   308→              </div>
   309→   309→
   310→   310→              {/* probability meter + phase dial + amplitude readout */}
   311→   311→              <div className="flex w-full min-w-0 flex-1 flex-col gap-5">
   312→   312→                <div dir={isAr ? "rtl" : "ltr"}>
   313→   313→                  <p className="mb-2 font-mono text-[10px] tracking-[0.25em] text-quantum-subtle">
   314→   314→                    <Atom className="me-1.5 inline h-3.5 w-3.5 text-quantum-blue" />
   315→   315→                    {pg.stateLabel}
   316→   316→                  </p>
   317→   317→                  <p
   318→   318→                    dir="ltr"
   319→   319→                    className="rounded-lg border border-quantum-blue/15 bg-quantum-navy/70 px-3 py-2.5 text-center font-mono text-sm text-quantum-blue"
   320→   320→                  >
   321→   321→                    |ψ⟩ = {cformat(state.a)}|0⟩ + {cformat(state.b)}|1⟩
   322→   322→                  </p>
   323→   323→                </div>
   324→   324→
   325→   325→                {/* Born-rule probability bars */}
   326→   326→                <div className="space-y-3" role="img" aria-label={`${pg.prob0Label} ${(p0 * 100).toFixed(0)}%, ${pg.prob1Label} ${(p1 * 100).toFixed(0)}%`}>
   327→   327→                  {(
   328→   328→                    [
   329→   329→                      { label: pg.prob0Label, p: p0, color: "#00D9FF" },

   355→   355→                {/* relative-phase dial — the invisible axis that makes quantum quantum */}

   358→   358→                    <circle
   359→   359→                      cx="28"
   360→   360→                      cy="28"
